def emit_event(event_type: str, **kwargs):
    """Emit SSE event to queue with timestamp"""
    event = {
        "type": event_type,
        "timestamp": datetime.now(timezone.utc).isoformat(),
        **kwargs
    }
    if hasattr(self, '_current_run_event_queue') and self._current_run_event_queue:
        try:
            self._current_run_event_queue.put_nowait(event)
        except Exception:
            pass
```

       def _calculate_tier_stats(self, run_state):
           """Calculate tier1/tier2/tier3 breakdown from results"""
           results = run_state.get("results", [])
           tier1_count = sum(1 for r in results if r.get("tier") == "tier1")
           tier2_count = sum(1 for r in results if r.get("tier") == "tier2")
           tier3_count = sum(1 for r in results if r.get("tier") == "tier3")
           total = len(results)

           return {
               "tier1": {"count": tier1_count, "percent": round(tier1_count / total * 100, 1) if total > 0 else 0.0},
               "tier2": {"count": tier2_count, "percent": round(tier2_count / total * 100, 1) if total > 0 else 0.0},
               "tier3": {"count": tier3_count, "percent": round(tier3_count / total * 100, 1) if total > 0 else 0.0},
               "total": total
           }
       ```

    3. Emit tier_stats event after each case_complete event:
       ```python

       yield {
           "type": "tier_stats",
           "stats": tier_stats,
           "timestamp": datetime.now(timezone.utc).isoformat()
       }
       ```

    4. Include tier_stats in init event for historical runs

    Implementation notes:
