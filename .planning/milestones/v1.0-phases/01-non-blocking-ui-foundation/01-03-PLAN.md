Eliminate blocking 3+ second startup (UI-01) by deferring heavy data loads:

1. **Split existing bootstrap() call** into fast and deferred sections:

```javascript
// In DOMContentLoaded or init():
async function initializeDashboard() {
  // FAST PATH: UI interactive in <500ms
  try {
    // Load only essential bootstrap data (tool configs, model list)
    const bootstrap = await fetch("/api/bootstrap").then(r => r.json());

    // Populate dropdowns immediately
    populateToolSelect(bootstrap.tools);
    populateValidationBackendSelect(bootstrap.backends);

    // Show ready state
    if (ui.homeStatus) ui.homeStatus.textContent = "ready";

    // Mark UI as interactive (measure this for UI-01 verification)
    console.timeEnd("ui-interactive"); // Started at top of <script>

  } catch (err) {
    console.error("Bootstrap failed:", err);
    if (ui.homeStatus) ui.homeStatus.textContent = "error";
    return;
  }

  // DEFERRED PATH: Load heavy data in background (doesn't block interaction)
  setTimeout(async () => {
    try {
      // Load run history (potentially large dataset)
      const runsData = await fetch("/api/runs").then(r => r.json());
      state.runs = runsData.runs || [];
      populateRunHistorySelect(state.runs);

      // Load loadouts
      const loadoutsData = await fetch("/api/loadouts").then(r => r.json());
      state.loadouts = loadoutsData.loadouts || [];
      populateLoadoutSelects(state.loadouts);

    } catch (err) {
      console.error("Deferred load failed:", err);
      // Non-fatal: UI still functional, just missing history/loadouts
    }
  }, 100); // Defer 100ms to ensure UI renders first
}

// Start timer at top of <script> tag for measurement
console.time("ui-interactive");

// Call on DOMContentLoaded
document.addEventListener("DOMContentLoaded", initializeDashboard);
```
